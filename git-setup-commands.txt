# Git 저장소 초기화 및 새 브랜치 생성 가이드
# https://github.com/ripplelover/GraphExtractors_revised.git 에 푸시하기

# 1. Git 저장소 초기화
git init

# 2. 리모트 저장소 추가
git remote add origin https://github.com/ripplelover/GraphExtractors_revised.git

# 3. 브랜치 이름을 정하세요 (예: feature-white-board)
# 새 브랜치 생성 및 체크아웃
git checkout -b feature-white-board

# 4. 모든 파일 추가
git add .

# 5. 커밋
git commit -m "Add white board feature"

# 6. 리모트 저장소에 푸시
git push -u origin feature-white-board

# 만약 리모트 저장소의 기존 내용을 먼저 가져와야 한다면:
# git pull origin main --allow-unrelated-histories
# 그 후 충돌 해결 후 다시 커밋하고 푸시
